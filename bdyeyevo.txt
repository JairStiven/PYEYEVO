CREATE DATABASE yeyevo;

USE yeyevo;

CREATE TABLE cliente (
    doc INT NOT NULL PRIMARY KEY,
    nombres VARCHAR(50) NOT NULL,
    apellidos VARCHAR(50) NOT NULL,
    correo VARCHAR(30) NOT NULL,
    telefono VARCHAR(15) NOT NULL,  
    direccion VARCHAR(30) NOT NULL
);

CREATE TABLE empleado (
    doc INT NOT NULL PRIMARY KEY,
    nombres VARCHAR(50) NOT NULL,
    apellidos VARCHAR(50) NOT NULL,
    cargo VARCHAR(20) NOT NULL,
    sueldo INT NOT NULL,
    correo VARCHAR(30) NOT NULL,
    telefono VARCHAR(15) NOT NULL,  
    direccion VARCHAR(30) NOT NULL
);

CREATE TABLE usuario (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    contraseña VARCHAR(50) NOT NULL,
    empleado_id INT UNIQUE,
    cliente_id INT UNIQUE,
    FOREIGN KEY (empleado_id) REFERENCES empleado(doc),
    FOREIGN KEY (cliente_id) REFERENCES cliente(doc)
);

CREATE TABLE forma_pago (
    codi INT NOT NULL PRIMARY KEY,
    desc_p VARCHAR(50) NOT NULL  
);

CREATE TABLE factura (
    num_factura INT NOT NULL PRIMARY KEY,
    fecha DATE NOT NULL,
    cod_p INT NOT NULL,
    doc_emp INT NOT NULL,
    doc_cli INT NOT NULL,
    FOREIGN KEY (cod_p) REFERENCES forma_pago(codi),
    FOREIGN KEY (doc_emp) REFERENCES empleado(doc),
    FOREIGN KEY (doc_cli) REFERENCES cliente(doc)
);

CREATE TABLE categoria (
    id INT NOT NULL PRIMARY KEY,
    nombre VARCHAR(20) NOT NULL
);

CREATE TABLE productos (
    id INT NOT NULL PRIMARY KEY,
    nombre VARCHAR(20) NOT NULL,
    valor_uni INT NOT NULL,
    cantidad_prod INT NOT NULL,
    id_cate INT NOT NULL,
    FOREIGN KEY (id_cate) REFERENCES categoria(id)
);

CREATE TABLE detalle_factura (
    id INT NOT NULL PRIMARY KEY,
    num_factura INT NOT NULL,
    cantidad INT NOT NULL,
    valor_venta INT NOT NULL,
    cod_prod INT NOT NULL,
    FOREIGN KEY (cod_prod) REFERENCES productos(id),
    FOREIGN KEY (num_factura) REFERENCES factura(num_factura)
);
